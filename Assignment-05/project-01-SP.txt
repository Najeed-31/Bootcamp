create PROCEDURE stp_AddVisit
    @PatientName varchar(100),
    @VisitTypeName varchar(50),
    @DoctorName varchar(100),
    @Description varchar(MAX),
    @DurationMinutes int
as
begin
    

    declare @PatientID int, @DoctorID int, @VisitTypeID int;
    declare @RatePerMinute decimal(10,2), @Fee decimal(10,2);
    declare @VisitDate datetime = getdate();

    BEGIN TRY
        BEGIN TRANSACTION;

        
        select @PatientID = PatientID 
        from Patients 
        where FullName = @PatientName;

        if @PatientID is null
        BEGIN
            insert into Patients (FullName)
            values 
            (@PatientName);
            set @PatientID = SCOPE_IDENTITY();
          
        END

        
        select @DoctorID = DoctorID
        from Doctors
        where FullName = @DoctorName;

        if @DoctorID is null
        BEGIN
            insert into Doctors (FullName)
            values
            (@DoctorName);
            SET @DoctorID = SCOPE_IDENTITY();
            
        END

        select @VisitTypeID = VisitTypeID
        from VisitTypes
        where TypeName = @VisitTypeName;

        if @VisitTypeID is null
        BEGIN
            RAISERROR('Invalid Visit Type.', 16, 1);
            ROLLBACK TRANSACTION;
            RETURN;
        END

        select @RatePerMinute = RatePerMinute
        from FeeSchedule
        where VisitTypeID = @VisitTypeID;

        set @Fee = @RatePerMinute * @DurationMinutes;

        insert into Visits 
        (
            PatientID, DoctorID, VisitTypeID, VisitDate, Description, DurationMinutes, Fee
        )
        values 
        (
            @PatientID, @DoctorID, @VisitTypeID, @VisitDate, @Description, @DurationMinutes, @Fee
        );

        insert into VisitHistory 
        (
            Action, PatientID, VisitDate, VisitTypeID, Description, DoctorID
        )
        values 
        (
            'Add', @PatientID, @VisitDate, @VisitTypeID, @Description, @DoctorID
        );

        COMMIT TRANSACTION;
        PRINT 'Visit added successfully.';

    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        PRINT ERROR_MESSAGE();
    END CATCH
END
GO

create PROCEDURE stp_UpdateVisit
    @PatientName varchar(100),
    @VisitTypeName varchar(50),
    @DoctorName varchar(100),
    @Description varchar(MAX),
    @DurationMinutes int
AS
BEGIN

    declare @PatientID int, @DoctorID int, @VisitTypeID int, @VisitID int;
    declare @RatePerMinute decimal(10,2), @Fee decimal(10,2);
    declare @VisitDate datetime = getdate();

    BEGIN TRY
        BEGIN TRANSACTION;

        select @PatientID = PatientID 
        from Patients
        where FullName = @PatientName;

        if @PatientID is null
        BEGIN
            RAISERROR('Patient does not exist.', 16, 1);
            ROLLBACK TRANSACTION;
            RETURN;
        END

        select @VisitID = VisitID
        from Visits
        where PatientID = @PatientID;

        if @VisitID is null
        BEGIN
            RAISERROR('Visit does not exist for the given patient.', 16, 1);
            ROLLBACK TRANSACTION;
            RETURN;
        END
        
        select @DoctorID = DoctorID
        from Doctors
        where FullName = @DoctorName;

        if @DoctorID is null
        BEGIN
            insert into Doctors (FullName)
            values 
            (@DoctorName);

            set @DoctorID = SCOPE_IDENTITY();
        END

        select @VisitTypeID = VisitTypeID
        from VisitTypes
        where TypeName = @VisitTypeName;

        if @VisitTypeID is null
        BEGIN
            RAISERROR('Invalid Visit Type.', 16, 1);
            ROLLBACK TRANSACTION;
            RETURN;
        END
        
        select @RatePerMinute = RatePerMinute
        from FeeSchedule
        where VisitTypeID = @VisitTypeID;

        set @Fee = @RatePerMinute * @DurationMinutes;

        update Visits
        set DoctorID = @DoctorID,
            VisitTypeID = @VisitTypeID,
            VisitDate = @VisitDate,
            Description = @Description,
            DurationMinutes = @DurationMinutes,
            Fee = @Fee
        where VisitID = @VisitID;

        INSERT INTO VisitHistory 
        (
            Action, PatientID, VisitDate, VisitTypeID, Description, DoctorID
        )
        VALUES 
        (
            'Update', @PatientID, @VisitDate, @VisitTypeID, @Description, @DoctorID
        );

        COMMIT TRANSACTION;
        PRINT 'Visit updated successfully.';

    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        PRINT ERROR_MESSAGE();
    END CATCH
END

create PROCEDURE stp_DeleteVisit
    @PatientName varchar(100),
    @VisitTypeName varchar(50),
    @DoctorName varchar(100),
    @DurationMinutes int
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @PatientID int, @DoctorID int, @VisitTypeID int, @VisitID int;
    DECLARE @VisitDate datetime = getdate();

    BEGIN TRY
        BEGIN TRANSACTION;

        select @PatientID = PatientID
        from Patients
        where FullName = @PatientName;

        if @PatientID is null
        BEGIN
            RAISERROR('Patient does not exist.', 16, 1);
            ROLLBACK TRANSACTION;
            RETURN;
        END

        select @VisitTypeID = VisitTypeID
        from VisitTypes
        where TypeName = @VisitTypeName;

        if @VisitTypeID is null
        BEGIN
            RAISERROR('Invalid Visit Type.', 16, 1);
            ROLLBACK TRANSACTION;
            RETURN;
        END

        select @DoctorID = DoctorID
        from Doctors
        where FullName = @DoctorName;

        if @DoctorID is null
        BEGIN
            RAISERROR('Doctor does not exist.', 16, 1);
            ROLLBACK TRANSACTION;
            RETURN;
        END

        select top 1 @VisitID = VisitID, @VisitDate = VisitDate
        from Visits
        where PatientID = @PatientID and VisitTypeID = @VisitTypeID and DoctorID = @DoctorID and DurationMinutes = @DurationMinutes;

        if @VisitID is null
        BEGIN
            RAISERROR('Visit does not exist for the given details.', 16, 1);
            ROLLBACK TRANSACTION;
            RETURN;
        END

        delete from Visits
        where VisitID = @VisitID;

        insert into VisitHistory 
        (
            Action, PatientID, VisitDate, VisitTypeID, Description, DoctorID
        )
        VALUES 
        (
            'Delete', @PatientID, @VisitDate, @VisitTypeID, 
            'Visit deleted from system.', @DoctorID
        );

        COMMIT TRANSACTION;
        PRINT 'Visit deleted successfully.';

    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        PRINT ERROR_MESSAGE();
    END CATCH
END

create PROCEDURE stp_SearchByPatient
    @PatientName varchar(100)
as
BEGIN

    declare @PatientID int;

    BEGIN TRY
        BEGIN TRANSACTION;

        select @PatientID = PatientID
        from Patients
        where FullName = @PatientName;

        if @PatientID is null
        BEGIN
            RAISERROR('Patient does not exist.', 16, 1);
            ROLLBACK TRANSACTION;
            RETURN;
        END

        if not exists (
            select 1
            from Visits
            where PatientID = @PatientID
        )
        BEGIN
            RAISERROR('No visits found for this patient.', 16, 1);
            ROLLBACK TRANSACTION;
            RETURN;
        END

        SELECT 
            v.VisitID,
            p.FullName as PatientName,
            d.FullName as DoctorName,
            vt.TypeName as VisitType,
            v.VisitDate,
            v.Description,
            v.DurationMinutes,
            v.Fee
        from Visits v
        inner join Patients p on v.PatientID = p.PatientID
        inner join Doctors d on v.DoctorID = d.DoctorID
        inner join VisitTypes vt on v.VisitTypeID = vt.VisitTypeID
        where v.PatientID = @PatientID;

        insert into VisitHistory
        (
            Action, PatientID, VisitDate, VisitTypeID, Description, DoctorID
        )
        select 
            'Search By Patient',
            v.PatientID,
            v.VisitDate,
            v.VisitTypeID,
            v.Description,
            v.DoctorID
        from Visits v
        where v.PatientID = @PatientID;

        COMMIT TRANSACTION;
        PRINT 'Search completed successfully.';

    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        PRINT ERROR_MESSAGE();
    END CATCH
END
GO

    @PatientName = 'Najeed Mubasher';
